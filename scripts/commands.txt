# Handy manual checks against a server started with scripts/run.sh (127.0.0.1:8080).

lsof -nP -iTCP:8080 -sTCP:LISTEN

curl -i http://127.0.0.1:8080/
curl -i http://127.0.0.1:8080/index.html

# HEAD requests (no body)
curl -I http://127.0.0.1:8080/
curl -I http://127.0.0.1:8080/swift.png
curl -I http://127.0.0.1:8080/Tab1/index.html

# ETag and 304: take the ETag (strong, metadata-based) from the first response
ET=$(curl -sI http://127.0.0.1:8080/index.html | grep -i '^etag' | cut -d' ' -f2- | tr -d '\r')
curl -i http://127.0.0.1:8080/index.html -H "If-None-Match: $ET"

# Last-Modified and If-Modified-Since
LM=$(curl -sI http://127.0.0.1:8080/index.html | grep -i '^last-modified' | cut -d' ' -f2- | tr -d '\r')
curl -i http://127.0.0.1:8080/index.html -H "If-Modified-Since: $LM"

# Range request (206)
curl -i -r 0-99 http://127.0.0.1:8080/swift.png -o /dev/null -D -

# 405 with Allow, 403 for blocked paths and traversal
curl -i -X POST http://127.0.0.1:8080/
curl -i http://127.0.0.1:8080/.DS_Store
curl -i --path-as-is http://127.0.0.1:8080/../http-config.json

# Stop the server
scripts/sigterm.sh            # or scripts/sigint.sh; PORT=9000 scripts/sigterm.sh for another port

# Build and run
swift build -j 2
swift run -c debug SwiftWebServer --config Root/http-config.json
swift run -c release SwiftWebServer --config Root/http-config.json --port 9000
swift build --target ConfigFile
scripts/guarded.sh -t 900 -- swift test -j 2
scripts/guarded.sh -t 600 -- swift test -j 2 --filter ParityTests

find Sources -name "*.swift" -print0 | xargs -0 cat | wc -l
